cmake_minimum_required(VERSION 3.16)
project(cicd_app CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Main application
add_executable(app src/main.cpp)
target_compile_options(app PRIVATE -Wall -Wextra)

# Tests (only when not building inside Docker)
option(BUILD_TESTS "Build tests" ON)
if(BUILD_TESTS)
    find_package(GTest QUIET)
    if(GTest_FOUND)
        enable_testing()
        add_executable(unit_tests tests/test_processor.cpp)
        target_link_libraries(unit_tests GTest::GTest GTest::Main)
        add_test(NAME UnitTests COMMAND unit_tests)
    else()
        # Try pkg-config fallback
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(GTEST gtest_main)
        endif()
        if(GTEST_FOUND)
            enable_testing()
            add_executable(unit_tests tests/test_processor.cpp)
            target_include_directories(unit_tests PRIVATE ${GTEST_INCLUDE_DIRS})
            target_link_libraries(unit_tests ${GTEST_LIBRARIES} pthread)
            add_test(NAME UnitTests COMMAND unit_tests)
        else()
            message(WARNING "Google Test not found, skipping unit tests")
        endif()
    endif()
endif()
